```typescript
export async function fetchEmployees(): Promise<Employee[]> {
  const timeout = new Promise<never>((_, reject) =>
    setTimeout(() => reject(new Error('timeout')), 30000)
  );

  const primary = supabase
    .from('employees')
    .select(FIELDS)
    .eq('is_dismissed', false)
    .order('name', { ascending: true });

  const primaryResult = await Promise.race([primary, timeout]);

  if (primaryResult.error) throw primaryResult.error;
  return resolvePhotoUrls(formatDataForApp((primaryResult.data ?? []) as unknown as DbRow[]));
}
```

The `Promise.race([primary, timeout])` line still throws 'timeout' via the rejected timeout promise — that path is preserved. The deleted lines are: the `if (primaryResult.error.message === 'timeout') throw` block (now redundant — any error throws), the `console.warn(...)` call, and the entire fallback block that used `FIELDS_BASE`.

**Change 2 — replace fetchOrganizations (lines 165-179) with the single-path version:**

```typescript
export async function fetchOrganizations(): Promise<string[]> {
  const toNames = (rows: { organization: unknown }[]) =>
    [...new Set(rows.map((i) => i.organization).filter(Boolean))].sort() as string[];

  const { data, error } = await supabase
    .from('employees')
    .select('organization')
    .eq('is_dismissed', false);

  if (error) throw new Error(error.message);
  return toNames(data ?? []);
}
```
