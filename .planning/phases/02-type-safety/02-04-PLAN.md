```typescript
describe('updatePermitWithStatus', () => {
  it('succeeds on first try when no CHECK constraint error', async () => {
    const mockPermit = { id: 'p-1', status: 'Активен' };
    (supabase.single as jest.Mock).mockResolvedValueOnce({ data: mockPermit, error: null });
    const result = await updatePermitWithStatus('p-1', { status: 'Активен' }, ['Активен', 'Просрочен']);
    expect(result).toEqual(mockPermit);
  });

  it('retries with next candidate on CHECK constraint error (23514)', async () => {
    const checkError = { code: '23514', message: 'Check constraint violated' };
    const mockPermit = { id: 'p-1', status: 'Просрочен' };
    // First attempt fails with CHECK error, second succeeds
    (supabase.single as jest.Mock)
      .mockResolvedValueOnce({ data: null, error: checkError })
      .mockResolvedValueOnce({ data: mockPermit, error: null });

    const result = await updatePermitWithStatus('p-1', { status: 'Активен' }, ['Активен', 'Просрочен']);
    expect(result).toEqual(mockPermit);
  });

  it('throws when all candidates are exhausted', async () => {
    const checkError = { code: '23514', message: 'Check constraint violated' };
    (supabase.single as jest.Mock)
      .mockResolvedValue({ data: null, error: checkError }); // always fails

    await expect(
      updatePermitWithStatus('p-1', { status: 'Активен' }, ['Активен'])
    ).rejects.toBeTruthy();
  });
});
```
